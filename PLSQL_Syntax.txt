03. VARIABLE DECLARATIONS, %TYPE, %ROWTYPE
--------------------------------------------------------------------------------
DECLARE
  v_id NUMBER := 101;
  v_name plx_emp.emp_name%TYPE;
  v_row plx_emp%ROWTYPE;
BEGIN
  SELECT * INTO v_row FROM plx_emp WHERE emp_id = v_id;
  v_name := v_row.emp_name;
  DBMS_OUTPUT.PUT_LINE(v_name);
END;
/

--------------------------------------------------------------------------------
04. IF / ELSIF / ELSE
--------------------------------------------------------------------------------
DECLARE
  v_marks NUMBER := 82;
BEGIN
  IF v_marks >= 90 THEN
    DBMS_OUTPUT.PUT_LINE('A');
  ELSIF v_marks >= 75 THEN
    DBMS_OUTPUT.PUT_LINE('B');
  ELSE
    DBMS_OUTPUT.PUT_LINE('C');
  END IF;
END;
/

--------------------------------------------------------------------------------
05. LOOP / WHILE / FOR
--------------------------------------------------------------------------------
BEGIN
  FOR i IN 1..3 LOOP
    DBMS_OUTPUT.PUT_LINE('FOR = ' || i);
  END LOOP;
END;
/

DECLARE
  i NUMBER := 1;
BEGIN
  WHILE i <= 3 LOOP
    DBMS_OUTPUT.PUT_LINE('WHILE = ' || i);
    i := i + 1;
  END LOOP;
END;
/

--------------------------------------------------------------------------------
06. SELECT INTO + EXCEPTION
--------------------------------------------------------------------------------
DECLARE
  v_salary NUMBER;
BEGIN
  SELECT salary INTO v_salary FROM plx_emp WHERE emp_id = 101;
  DBMS_OUTPUT.PUT_LINE(v_salary);
EXCEPTION
  WHEN NO_DATA_FOUND THEN DBMS_OUTPUT.PUT_LINE('No matching employee');
  WHEN TOO_MANY_ROWS THEN DBMS_OUTPUT.PUT_LINE('More than one row');
END;
/

10. PROCEDURE: INSERT / UPDATE / DELETE

-- Assumes PLX_EMP exists. Use a new emp_id if inserting twice.
INSERT INTO plx_emp(emp_id, emp_name, dept_id, salary)
VALUES (111, 'New Hire', 10, 42000);
UPDATE plx_emp SET salary = 44000 WHERE emp_id = 111;
DELETE FROM plx_emp WHERE emp_id = 111;
ROLLBACK;

12. CREATE FUNCTION RETURN VARCHAR2
--------------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION plx_fn_pass(p_mark IN NUMBER)
RETURN VARCHAR2
AS
BEGIN
  IF p_mark >= 50 THEN
    RETURN 'PASS';
  ELSE
    RETURN 'FAIL';
  END IF;
END;
/
SELECT plx_fn_pass(68) AS result FROM dual;

--------------------------------------------------------------------------------
13. FUNCTION SELECT INTO
--------------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION plx_fn_salary(p_id IN NUMBER)
RETURN NUMBER
AS
  v_salary NUMBER;
BEGIN
  SELECT salary INTO v_salary FROM plx_emp WHERE emp_id = p_id;
  RETURN v_salary;
EXCEPTION
  WHEN NO_DATA_FOUND THEN
    RAISE_APPLICATION_ERROR(-20201, 'Employee not found');
END;
/
SELECT plx_fn_salary(101) AS salary FROM dual;

15. PROCEDURE VS FUNCTION (EXAM DIFFERENCE)
--------------------------------------------------------------------------------
PROCEDURE                       FUNCTION
-----------------------------------------------------------------------
CREATE PROCEDURE                CREATE FUNCTION
Called using EXEC proc(...)     SELECT func(...) FROM dual;
Can have IN/OUT/IN OUT          Typically IN inputs for SQL-callable functions
No mandatory RETURN value      RETURN datatype + RETURN expression required
Often used for actions / DML   Often used for computing/returning a value

16. CREATE BEFORE INSERT TRIGGER
--------------------------------------------------------------------------------
CREATE OR REPLACE TRIGGER plx_tr_example_before_insert
BEFORE INSERT ON plx_emp
FOR EACH ROW
BEGIN
  IF :NEW.salary < 10000 THEN
    RAISE_APPLICATION_ERROR(-20202, 'Salary too low');
  END IF;
END;
/
-- This trigger fires when an INSERT occurs. No EXEC trigger_name.

--------------------------------------------------------------------------------
17. CREATE AFTER INSERT TRIGGER
--------------------------------------------------------------------------------
CREATE OR REPLACE TRIGGER plx_tr_example_after_insert
AFTER INSERT ON plx_emp
FOR EACH ROW
BEGIN
  INSERT INTO plx_audit(audit_id, emp_id, action_type, old_salary,
                        new_salary, logged_at)
  VALUES (plx_audit_seq.NEXTVAL, :NEW.emp_id, 'EXAMPLE_INSERT', NULL,
          :NEW.salary, SYSTIMESTAMP);
END;
/

19. CREATE AFTER UPDATE TRIGGER + :OLD/:NEW
--------------------------------------------------------------------------------
CREATE OR REPLACE TRIGGER plx_tr_example_after_update
AFTER UPDATE OF salary ON plx_emp
FOR EACH ROW
BEGIN
  DBMS_OUTPUT.PUT_LINE('Old=' || :OLD.salary || ', New=' || :NEW.salary);
END;
/

21. ROW-LEVEL VS STATEMENT-LEVEL TRIGGER
--------------------------------------------------------------------------------
-- Statement-level: NO FOR EACH ROW.
CREATE OR REPLACE TRIGGER plx_tr_example_statement
AFTER UPDATE OF status ON plx_emp
BEGIN
  DBMS_OUTPUT.PUT_LINE('One UPDATE statement completed');
END;
/

-- Row-level: add FOR EACH ROW to run once per affected row.
-- The example in previous section uses FOR EACH ROW.

--------------------------------------------------------------------------------
22. MULTI-EVENT TRIGGER + INSERTING/UPDATING/DELETING
--------------------------------------------------------------------------------
CREATE OR REPLACE TRIGGER plx_tr_example_events
AFTER INSERT OR UPDATE OR DELETE ON plx_emp
BEGIN
  IF INSERTING THEN
    DBMS_OUTPUT.PUT_LINE('An INSERT statement occurred');
  ELSIF UPDATING THEN
    DBMS_OUTPUT.PUT_LINE('An UPDATE statement occurred');
  ELSIF DELETING THEN
    DBMS_OUTPUT.PUT_LINE('A DELETE statement occurred');
  END IF;
END;
/

--------------------------------------------------------------------------------
23. WHEN CONDITION / UPDATING(column)
--------------------------------------------------------------------------------
CREATE OR REPLACE TRIGGER plx_tr_example_when
AFTER UPDATE OF salary ON plx_emp
FOR EACH ROW
WHEN (NEW.salary > OLD.salary)
BEGIN
  DBMS_OUTPUT.PUT_LINE('Increase only: ' || :NEW.emp_id);
END;
/

-- WHEN: NEW / OLD have NO colon; inside BEGIN, use :NEW / :OLD.
-- UPDATING('SALARY') is useful in multi-event row triggers.

--------------------------------------------------------------------------------
24. COMPOUND TRIGGER
--------------------------------------------------------------------------------
CREATE OR REPLACE TRIGGER plx_tr_example_compound
FOR UPDATE OF salary ON plx_emp
COMPOUND TRIGGER
  g_count NUMBER := 0;
  AFTER EACH ROW IS
  BEGIN
    g_count := g_count + 1;
  END AFTER EACH ROW;
  AFTER STATEMENT IS
  BEGIN
    DBMS_OUTPUT.PUT_LINE('Updated rows: ' || g_count);
  END AFTER STATEMENT;
END plx_tr_example_compound;
/

25. EXCEPTION, RAISE_APPLICATION_ERROR
--------------------------------------------------------------------------------
BEGIN
  IF 5 < 10 THEN
    RAISE_APPLICATION_ERROR(-20203, 'Demo custom error');
  END IF;
EXCEPTION
  WHEN OTHERS THEN
    DBMS_OUTPUT.PUT_LINE('Caught: ' || SQLERRM);
END;
/

-- User-defined error codes commonly use -20000 through -20999.

26. DBMS_OUTPUT, SQL%ROWCOUNT

27. CURSOR FOR LOOP IN PROCEDURE
--------------------------------------------------------------------------------
CREATE OR REPLACE PROCEDURE plx_pr_example_cursor
AS
BEGIN
  FOR r IN (SELECT emp_name, salary FROM plx_emp ORDER BY emp_id) LOOP
    DBMS_OUTPUT.PUT_LINE(r.emp_name || ': ' || r.salary);
  END LOOP;
END;
/
EXEC plx_pr_example_cursor
