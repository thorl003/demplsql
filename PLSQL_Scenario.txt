PROMPT === 00: CREATE TABLES ===
CREATE TABLE plx_dept (
  dept_id NUMBER PRIMARY KEY,
  dept_name VARCHAR2(40) NOT NULL UNIQUE
);

CREATE TABLE plx_emp (
  emp_id NUMBER PRIMARY KEY,
  emp_name VARCHAR2(50) NOT NULL,
  dept_id NUMBER REFERENCES plx_dept(dept_id),
  salary NUMBER(10,2) NOT NULL,
  hire_date DATE DEFAULT SYSDATE NOT NULL,
  status VARCHAR2(12) DEFAULT 'ACTIVE' NOT NULL,
  updated_at DATE DEFAULT SYSDATE NOT NULL,
  CONSTRAINT plx_emp_salary_ck CHECK (salary >= 0)
);

CREATE TABLE plx_audit (
  audit_id NUMBER PRIMARY KEY,
  emp_id NUMBER,
  action_type VARCHAR2(30),
  old_salary NUMBER(10,2),
  new_salary NUMBER(10,2),
  logged_at TIMESTAMP
);

CREATE SEQUENCE plx_audit_seq START WITH 1 INCREMENT BY 1;

CREATE TABLE plx_events (
  event_id NUMBER PRIMARY KEY,
  event_label VARCHAR2(120),
  event_at TIMESTAMP DEFAULT SYSTIMESTAMP
);

CREATE SEQUENCE plx_event_seq START WITH 1 INCREMENT BY 1;

CREATE TABLE plx_bonus (
  bonus_id NUMBER PRIMARY KEY,
  emp_id NUMBER REFERENCES plx_emp(emp_id),
  bonus_amount NUMBER(10,2),
  created_at DATE DEFAULT SYSDATE
);

CREATE SEQUENCE plx_bonus_seq START WITH 1 INCREMENT BY 1;

PROBLEM 03: PROCEDURE USING SELECT INTO + NO_DATA_FOUND
--------------------------------------------------------------------------------
Goal       : Procedure using select into + no_data_found.
Key Ops    : PROCEDURE, SQL*Plus, PL/SQL

PROMPT === PROBLEM 03: PROCEDURE WITH SELECT INTO ===
CREATE OR REPLACE PROCEDURE plx_pr_show_employee(p_id IN NUMBER)
AS
  v_name plx_emp.emp_name%TYPE;
  v_salary plx_emp.salary%TYPE;
BEGIN
  SELECT emp_name, salary INTO v_name, v_salary
  FROM plx_emp WHERE emp_id = p_id;
  DBMS_OUTPUT.PUT_LINE('Employee: ' || v_name || ', Salary: ' || v_salary);
EXCEPTION
  WHEN NO_DATA_FOUND THEN
    DBMS_OUTPUT.PUT_LINE('No employee for ID ' || p_id);
END;
/
EXEC plx_pr_show_employee(101)
EXEC plx_pr_show_employee(999)

PROBLEM 04: PROCEDURE TO UPDATE EMPLOYEE SALARY
--------------------------------------------------------------------------------
Goal       : Procedure to update employee salary.
Key Ops    : PROCEDURE, SQL*Plus, PL/SQL

PROMPT === PROBLEM 04: PROCEDURE THAT UPDATES SALARY ===
CREATE OR REPLACE PROCEDURE plx_pr_raise_salary(
  p_id IN NUMBER, p_raise IN NUMBER
)
AS
BEGIN
  IF p_raise <= 0 OR p_raise IS NULL THEN
    RAISE_APPLICATION_ERROR(-20001, 'Raise must be positive');
  END IF;
  UPDATE plx_emp SET salary = salary + p_raise WHERE emp_id = p_id;
  IF SQL%ROWCOUNT = 0 THEN
    RAISE_APPLICATION_ERROR(-20002, 'Employee not found');
  END IF;
  DBMS_OUTPUT.PUT_LINE('Updated salary for ID ' || p_id);
END;
/
EXEC plx_pr_raise_salary(101, 1000)
SELECT emp_id, salary FROM plx_emp WHERE emp_id = 101;
COMMIT;

AS

PROBLEM 12: FUNCTION TO COMPUTE GRADE
--------------------------------------------------------------------------------
Goal       : Function to compute grade.
Key Ops    : FUNCTION, SQL*Plus, PL/SQL

PROMPT === PROBLEM 12: GRADE FUNCTION ===
CREATE OR REPLACE FUNCTION plx_fn_grade(p_mark IN NUMBER)
RETURN VARCHAR2
AS
BEGIN
  IF p_mark IS NULL OR p_mark < 0 OR p_mark > 100 THEN
    RETURN 'INVALID';
  ELSIF p_mark >= 90 THEN
    RETURN 'A';
  ELSIF p_mark >= 75 THEN
    RETURN 'B';
  ELSIF p_mark >= 50 THEN
    RETURN 'C';
  ELSE
    RETURN 'FAIL';
  END IF;
END;
/
SELECT plx_fn_grade(92) AS grade_92,
       plx_fn_grade(48) AS grade_48 FROM dual;

PROBLEM 13: FUNCTION TO RETURN ANNUAL SALARY
--------------------------------------------------------------------------------

PROMPT === PROBLEM 13: ANNUAL SALARY USING SELECT INTO ===
CREATE OR REPLACE FUNCTION plx_fn_annual_salary(p_id IN NUMBER)
RETURN NUMBER

  v_monthly plx_emp.salary%TYPE;

  WHEN NO_DATA_FOUND THEN
    RAISE_APPLICATION_ERROR(-20013, 'Employee ID does not exist');
END;

PROBLEM 15: FUNCTION TO CALCULATE FACTORIAL
--------------------------------------------------------------------------------
Goal       : Function to calculate factorial.
Key Ops    : FUNCTION, SQL*Plus, PL/SQL

PROMPT === PROBLEM 15: FACTORIAL USING LOOP ===
CREATE OR REPLACE FUNCTION plx_fn_factorial(p_n IN NUMBER)
RETURN NUMBER
AS
  v_result NUMBER := 1;
BEGIN
  IF p_n IS NULL OR p_n < 0 OR p_n > 20 OR p_n != TRUNC(p_n) THEN
    RAISE_APPLICATION_ERROR(-20015, 'Use an integer between 0 and 20');
  END IF;
  FOR i IN 1..p_n LOOP
    v_result := v_result * i;
  END LOOP;
  RETURN v_result;
END;
/
SELECT plx_fn_factorial(5) AS factorial_5 FROM dual;

PROBLEM 17: FUNCTION TO COMPUTE BONUS PERCENTAGE
--------------------------------------------------------------------------------
Goal       : Function to compute bonus percentage.

PROMPT === PROBLEM 17: SIMPLE BONUS FUNCTION ===
CREATE OR REPLACE FUNCTION plx_fn_bonus(p_salary IN NUMBER, p_rate IN NUMBER)
RETURN NUMBER
AS
BEGIN
  IF p_rate < 0 OR p_rate > 100 OR p_rate IS NULL THEN
    RAISE_APPLICATION_ERROR(-20017, 'Rate must be in range 0 to 100');
  END IF;
  RETURN ROUND(p_salary * p_rate / 100, 2);
END;
/

PROBLEM 26: COMPOUND TRIGGER FOR UPDATED-ROW COUNT
--------------------------------------------------------------------------------
Goal       : Compound trigger for updated-row count.
Key Ops    : TRIGGER, SQL*Plus, PL/SQL

PROMPT === PROBLEM 26: COMPOUND TRIGGER (ADVANCED, COUNTS ROWS) ===
CREATE OR REPLACE TRIGGER plx_tr_compound_count
FOR UPDATE OF salary ON plx_emp
COMPOUND TRIGGER
  g_rows NUMBER := 0;
  AFTER EACH ROW IS
  BEGIN
    g_rows := g_rows + 1;
  END AFTER EACH ROW;
  AFTER STATEMENT IS
  BEGIN
    DBMS_OUTPUT.PUT_LINE('Compound trigger updated rows = ' || g_rows);
  END AFTER STATEMENT;
END plx_tr_compound_count;
/
UPDATE plx_emp SET salary = salary + 1 WHERE dept_id = 10;
COMMIT;
